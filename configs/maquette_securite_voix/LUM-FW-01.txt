no dhcpd enable inside

 nameif outside
 security-level 0
 ip address 203.0.113.4 255.255.255.248
 ipv6 address 2001:db8:13:f000::4/64
 no shutdown
exit
interface GigabitEthernet1/2
 nameif inside
 security-level 100
 ip address 10.13.253.1 255.255.255.248
 ipv6 address 2001:db8:13:fd00::1/64
 no shutdown
exit
interface GigabitEthernet1/3
 nameif dmz
 security-level 50
 ip address 10.13.210.1 255.255.255.0
 ipv6 address 2001:db8:13:210::1/64
 no shutdown
exit
! IPv6 : lignes eventuellement refusees selon la version de PT (sans incidence sur l'IPv4)
ipv6 route outside ::/0 2001:db8:13:f000::2
ipv6 route inside 2001:db8:13::/48 2001:db8:13:fd00::2
ipv6 route inside 2001:db8:4::/48 2001:db8:13:fd00::2
route outside 0.0.0.0 0.0.0.0 203.0.113.1 1
route inside 10.13.0.0 255.255.0.0 10.13.253.4 1
route inside 10.4.0.0 255.255.0.0 10.13.253.4 1
object network LAN-LUMINY
 subnet 10.13.0.0 255.255.0.0
 nat (inside,outside) dynamic interface
object network LAN-DIGNE-SECOURS
 subnet 10.4.0.0 255.255.0.0
 nat (inside,outside) dynamic interface
object network DMZ-WWW
 host 10.13.210.10
 nat (dmz,outside) static 203.0.113.17
object network DMZ-DNS
 host 10.13.210.30
 nat (dmz,outside) static 203.0.113.19
access-list OUTSIDE-IN extended permit tcp any host 10.13.210.10 eq www
access-list OUTSIDE-IN extended permit tcp any host 10.13.210.10 eq 443
access-list OUTSIDE-IN extended permit udp any host 10.13.210.30 eq domain
access-list OUTSIDE-IN extended deny ip any any
access-group OUTSIDE-IN in interface outside
access-list DMZ-IN extended permit udp 10.13.210.0 255.255.255.0 host 10.13.200.50 eq ntp
access-list DMZ-IN extended permit udp 10.13.210.0 255.255.255.0 host 10.13.200.50 eq syslog
access-list DMZ-IN extended deny ip 10.13.210.0 255.255.255.0 10.13.0.0 255.255.0.0
access-list DMZ-IN extended deny ip 10.13.210.0 255.255.255.0 10.4.0.0 255.255.0.0
access-list DMZ-IN extended permit tcp 10.13.210.0 255.255.255.0 any eq www
access-list DMZ-IN extended permit tcp 10.13.210.0 255.255.255.0 any eq 443
access-list DMZ-IN extended permit udp 10.13.210.0 255.255.255.0 any eq domain
access-group DMZ-IN in interface dmz
class-map inspection_default
 match default-inspection-traffic
exit
policy-map global_policy
 class inspection_default
  inspect icmp
  inspect dns
  inspect http
  inspect ftp
exit
exit
service-policy global_policy global
end
write memory
