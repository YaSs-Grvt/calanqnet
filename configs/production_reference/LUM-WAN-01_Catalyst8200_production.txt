! =====================================================================
! CALANQ'NET - Configuration de reference PRODUCTION - LUM-WAN-01 (CE MPLS, Catalyst 8200)
! QoS a 4 classes alignee sur le contrat operateur + BGP PE-CE. Partie de Yacine Ben Omrane (Topic 7)
! =====================================================================
hostname LUM-WAN-01
class-map match-any CM-VOIX
 match dscp ef
class-map match-any CM-VISIO
 match dscp af41 af42
class-map match-any CM-SIGNAL
 match dscp cs3
class-map match-any CM-CRITIQUE
 match dscp af31
!
policy-map PM-WAN-ENFANT
 class CM-VOIX
  priority percent 20
 class CM-SIGNAL
  bandwidth percent 5
 class CM-VISIO
  bandwidth percent 25
 class CM-CRITIQUE
  bandwidth percent 25
  random-detect dscp-based
 class class-default
  fair-queue
  random-detect
policy-map PM-WAN-PARENT
 class class-default
  shape average 950000000
  service-policy PM-WAN-ENFANT
!
interface GigabitEthernet0/0/0
 description Acces MPLS L3VPN 1 Gb/s (PE Marseille)
 ip address 10.255.13.2 255.255.255.252
 ipv6 address 2001:DB8:FFFF:13::2/64
 service-policy output PM-WAN-PARENT
 bfd interval 300 min_rx 300 multiplier 3
!
ip prefix-list PL-LUM-OUT seq 5 permit 10.13.0.0/16
ip prefix-list PL-DIG-IN seq 5 permit 10.4.0.0/16
route-map RM-PE-OUT permit 10
 match ip address prefix-list PL-LUM-OUT
route-map RM-PE-IN permit 10
 match ip address prefix-list PL-DIG-IN
!
router bgp 65013
 bgp router-id 10.13.255.7
 neighbor 10.255.13.1 remote-as 64510
 neighbor 10.255.13.1 fall-over bfd
 neighbor 2001:DB8:FFFF:13::1 remote-as 64510
 address-family ipv4
  network 10.13.0.0 mask 255.255.0.0
  neighbor 10.255.13.1 activate
  neighbor 10.255.13.1 route-map RM-PE-IN in
  neighbor 10.255.13.1 route-map RM-PE-OUT out
 exit-address-family
 address-family ipv6
  network 2001:DB8:13::/48
  neighbor 2001:DB8:FFFF:13::1 activate
 exit-address-family
!
router ospf 1
 redistribute bgp 65013 subnets route-map RM-PE-IN
!
! Secours : en cas de perte du MPLS, la route 10.4.0.0/16 apprise via le tunnel IPsec des
! pare-feu (distance administrative superieure) prend le relais automatiquement.
