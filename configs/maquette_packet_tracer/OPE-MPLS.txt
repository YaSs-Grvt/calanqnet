enable
configure terminal
hostname OPE-MPLS
no ip domain-lookup
ip domain-name iutrt.lan
enable secret Ch@ngeMe-Lab!
username admin privilege 15 secret Ch@ngeMe-Lab!
service password-encryption
banner motd #Acces reserve au personnel autorise - CALANQ'NET / IUT R&T Luminy. Toute connexion est journalisee.#
ipv6 unicast-routing
interface GigabitEthernet0/0/0
 description PE Marseille - CE Luminy
 ip address 10.255.13.1 255.255.255.252
 ipv6 address 2001:db8:ffff:13::1/64
 no shutdown
exit
interface GigabitEthernet0/0/1
 description PE Digne - CE Digne
 ip address 10.255.4.1 255.255.255.252
 ipv6 address 2001:db8:ffff:4ff::1/64
 no shutdown
exit
router bgp 64510
 bgp router-id 10.255.255.1
 neighbor 10.255.13.2 remote-as 65013
 neighbor 10.255.4.2 remote-as 65004
exit
ipv6 route 2001:db8:13::/48 2001:db8:ffff:13::2
ipv6 route 2001:db8:4::/48 2001:db8:ffff:4ff::2
end
write memory
