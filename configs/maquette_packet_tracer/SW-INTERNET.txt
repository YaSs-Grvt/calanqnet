enable
configure terminal
hostname SW-INTERNET
no ip domain-lookup
ip domain-name iutrt.lan
enable secret Ch@ngeMe-Lab!
username admin privilege 15 secret Ch@ngeMe-Lab!
service password-encryption
banner motd #Acces reserve au personnel autorise - CALANQ'NET / IUT R&T Luminy. Toute connexion est journalisee.#
vtp mode transparent
spanning-tree mode rapid-pvst
interface range FastEthernet0/1 - 24 , GigabitEthernet0/1 - 2
 spanning-tree portfast
 no shutdown
exit
ip ssh version 2
crypto key generate rsa general-keys modulus 2048
line con 0
 logging synchronous
 exec-timeout 10 0
line vty 0 15
 login local
 transport input ssh
 exec-timeout 10 0
exit
ntp server 10.13.200.50
logging host 10.13.200.50
logging trap informational
snmp-server community Cq-RO-Zabbix RO
end
write memory
